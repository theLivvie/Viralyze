   253→   253→  const [compareOpen, setCompareOpen] = useState(false);
   254→   254→
   255→   255→  const fetchLibrary = useCallback(async () => {
   256→   256→    if (!user?.id) {
   257→   257→      setLoading(false);
   258→   258→      return;
   259→   259→    }
   260→   260→    try {
   261→   261→      setLoading(true);

   264→   264→        const data = await res.json();
   265→   265→        setAnalyses(data);
   266→   266→        setSavedAnalyses(data);
   267→   267→      }
   268→   268→    } catch {
   269→   269→      // Silently fail — use whatever is in local state
   270→   270→    } finally {
   271→   271→      setLoading(false);
   272→   272→    }
   273→   273→  }, [user?.id, setSavedAnalyses]);
