   125→// --- Animated Counter Component ---
   126→function AnimatedCounter({ value, duration = 1.2 }: { value: number; duration?: number }) {
   127→  const [display, setDisplay] = useState(0);
   128→  const ref = useRef(false);
   129→
   130→  useEffect(() => {
   131→    if (ref.current) return;
   132→    ref.current = true;
   133→    const mv = useMotionValue(0);
   134→    const unsub = useTransform(mv, (v) => setDisplay(Math.round(v))).on('change', () => {});
   135→    const ctrl = animate(mv, value, { duration, ease: 'easeOut' });
   136→    return () => {
   137→      ctrl.stop();
   138→      unsub();
   139→    };
   140→  // eslint-disable-next-line react-hooks/exhaustive-deps
   141→  }, [value, duration]);
   142→
   143→  return <>{display}</>;
   144→}
