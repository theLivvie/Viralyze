   243→export default function LibraryView() {
   244→  const { setSavedAnalyses, removeSavedAnalysis, setCurrentAnalysis, setCurrentView, user } = useAppStore();
   245→  const [analyses, setAnalyses] = useState<SavedAnalysis[]>([]);
   246→  const [loading, setLoading] = useState(true);
   247→  const [search, setSearch] = useState('');
   248→  const [platformFilter, setPlatformFilter] = useState<string>('all');
   249→  const [sort, setSort] = useState<string>('newest');
   250→  const [deleting, setDeleting] = useState<string | null>(null);
   251→  const [compareMode, setCompareMode] = useState(false);
   252→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   253→  const [compareOpen, setCompareOpen] = useState(false);
   254→
   255→  const fetchLibrary = useCallback(async () => {
   256→    if (!user?.id) {
   257→      setLoading(false);
   258→      return;
   259→    }
   260→    try {
   261→      setLoading(true);
   262→      const res = await fetch(`/api/library?userId=${user.id}`);
   263→      if (res.ok) {
   264→        const data = await res.json();
   265→        setAnalyses(data);
   266→        setSavedAnalyses(data);
   267→      }
   268→    } catch {
   269→      // Silently fail — use whatever is in local state
   270→    } finally {
   271→      setLoading(false);
   272→    }
   273→  }, [user?.id, setSavedAnalyses]);
   274→
   275→  useEffect(() => {
   276→    fetchLibrary();
   277→  }, [fetchLibrary]);
   278→
   279→  const filtered = analyses
   280→    .filter((a) => {
   281→      if (search && !a.title.toLowerCase().includes(search.toLowerCase())) return false;
   282→      if (platformFilter !== 'all' && a.platform !== platformFilter) return false;
   283→      return true;
   284→    })
   285→    .sort((a, b) => {
   286→      if (sort === 'newest') return new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
   287→      if (sort === 'oldest') return new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime();
   288→      if (sort === 'score-high') return b.overallScore - a.overallScore;
   289→      if (sort === 'score-low') return a.overallScore - b.overallScore;
   290→      return 0;
   291→    });
   292→
   293→  const handleDelete = async (e: React.MouseEvent, id: string) => {
   294→    e.stopPropagation();
   295→    if (!user?.id) return;
   296→    setDeleting(id);
   297→    try {
   298→      const res = await fetch(`/api/library?id=${id}&userId=${user.id}`, { method: 'DELETE' });
   299→      if (res.ok) {
   300→        setAnalyses((prev) => prev.filter((a) => a.id !== id));
   301→        removeSavedAnalysis(id);
   302→        setSelectedIds((prev) => { const next = new Set(prev); next.delete(id); return next; });
   303→        toast.success('Removed from library');
   304→      } else {
   305→        toast.error('Failed to delete analysis');
   306→      }
   307→    } catch {
   308→      toast.error('Failed to delete analysis');
   309→    } finally {
   310→      setDeleting(null);
   311→    }
   312→  };
   313→
   314→  const [viewingId, setViewingId] = useState<string | null>(null);
   315→
   316→  const handleCardClick = async (id: string) => {
   317→    if (compareMode) {
   318→      setSelectedIds((prev) => {
   319→        const next = new Set(prev);
   320→        if (next.has(id)) {
   321→          next.delete(id);
   322→        } else if (next.size < 3) {
   323→          next.add(id);
   324→        } else {
   325→          toast.info('You can compare up to 3 items');
   326→          return prev;
   327→        }
   328→        return next;
   329→      });
   330→    } else {
   331→      try {
   332→        setViewingId(id);
   333→        const res = await fetch(`/api/library?id=${id}`);
   334→        if (res.ok) {
   335→          const data = await res.json();
   336→          setCurrentAnalysis(data);
   337→          setCurrentView('analysis');
   338→        } else {
   339→          toast.error('Failed to load analysis');
   340→        }
   341→      } catch {
   342→        toast.error('Failed to load analysis');
   343→      } finally {
   344→        setViewingId(null);
   345→      }
   346→    }
   347→  };
   348→
   349→  const selectedAnalyses = analyses.filter((a) => selectedIds.has(a.id));
   350→
   351→  const handleCompare = () => {
   352→    if (selectedIds.size >= 2) {
   353→      setCompareOpen(true);
   354→    }
   355→  };
   356→
   357→  const exitCompareMode = () => {
   358→    setCompareMode(false);
   359→    setSelectedIds(new Set());
   360→  };
   361→
   362→  const isFilteredEmpty = !loading && filtered.length === 0 && analyses.length > 0;
   363→  const isEmpty = !loading && analyses.length === 0;
   364→
   365→  return (
   366→    <motion.div
   367→      variants={container}
   368→      initial="hidden"
   369→      animate="show"
   370→      className="flex flex-col gap-4 max-w-4xl mx-auto"
   371→    >
   372→      {/* Header with count badge */}
   373→      <motion.div variants={item} className="flex items-center gap-3">
   374→        <div className="flex flex-col">
   375→          <div className="flex items-center gap-2.5">
   376→            <h2 className="text-2xl md:text-3xl font-bold text-viralyze-white">
   377→              Content Library
   378→            </h2>
   379→            {!loading && analyses.length > 0 && (
   380→              <Badge
   381→                variant="outline"
   382→                className="text-xs border-wine-accent/30 text-wine-accent bg-wine-accent/10 tabular-nums"
   383→              >
   384→                {analyses.length}
   385→              </Badge>
   386→            )}
   387→          </div>
   388→          <p className="text-viralyze-muted mt-1">
   389→            Your saved content analyses
   390→          </p>
   391→        </div>
   392→      </motion.div>
   393→
   394→      {/* Filters */}
   395→      <motion.div variants={item} className="flex flex-col sm:flex-row gap-3">
   396→        <div className="relative flex-1">
   397→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-viralyze-muted" />
   398→          <Input
   399→            placeholder="Search analyses..."
   400→            value={search}
   401→            onChange={(e) => setSearch(e.target.value)}
   402→            className="pl-9 bg-white/[0.05] border-white/[0.08] text-viralyze-white placeholder:text-viralyze-muted/40 focus-visible:ring-wine-accent hover:border-white/[0.12] transition-colors"
   403→          />
   404→        </div>
   405→        <Select value={platformFilter} onValueChange={setPlatformFilter}>
   406→          <SelectTrigger className="w-full sm:w-40 bg-white/[0.05] border-white/[0.08] text-viralyze-white hover:border-white/[0.12] transition-colors">
   407→            <SelectValue placeholder="Platform" />
   408→          </SelectTrigger>
   409→          <SelectContent className="bg-viralyze-soft-black border-white/[0.08]">
   410→            <SelectItem value="all" className="text-viralyze-white">All Platforms</SelectItem>
   411→            <SelectItem value="instagram" className="text-viralyze-white">Instagram</SelectItem>
   412→            <SelectItem value="youtube" className="text-viralyze-white">YouTube</SelectItem>
   413→            <SelectItem value="tiktok" className="text-viralyze-white">TikTok</SelectItem>
   414→            <SelectItem value="x" className="text-viralyze-white">X</SelectItem>
   415→            <SelectItem value="linkedin" className="text-viralyze-white">LinkedIn</SelectItem>
   416→          </SelectContent>
   417→        </Select>
   418→        <Select value={sort} onValueChange={setSort}>
   419→          <SelectTrigger className="w-full sm:w-40 bg-white/[0.05] border-white/[0.08] text-viralyze-white hover:border-white/[0.12] transition-colors">
   420→            <SelectValue placeholder="Sort" />
   421→          </SelectTrigger>
   422→          <SelectContent className="bg-viralyze-soft-black border-white/[0.08]">
   423→            <SelectItem value="newest" className="text-viralyze-white">Newest First</SelectItem>
   424→            <SelectItem value="oldest" className="text-viralyze-white">Oldest First</SelectItem>
   425→            <SelectItem value="score-high" className="text-viralyze-white">Score: High to Low</SelectItem>
   426→            <SelectItem value="score-low" className="text-viralyze-white">Score: Low to High</SelectItem>
   427→          </SelectContent>
   428→        </Select>
   429→      </motion.div>
   430→
   431→      {/* Compare controls */}
   432→      {!loading && analyses.length > 1 && (
   433→        <motion.div variants={item} className="flex items-center gap-2">
   434→          <Button
   435→            variant={compareMode ? 'default' : 'outline'}
   436→            size="sm"
   437→            onClick={() => {
   438→              if (compareMode) {
   439→                exitCompareMode();
   440→              } else {
   441→                setCompareMode(true);
   442→              }
   443→            }}
   444→            className={cn(
   445→              'gap-1.5 transition-all',
   446→              compareMode
   447→                ? 'bg-wine-accent hover:bg-wine-accent/80 text-white border-wine-accent'
   448→                : 'border-white/[0.08] text-viralyze-white hover:bg-white/[0.05] hover:border-white/20'
   449→            )}
   450→          >
   451→            <GitCompareArrows className="h-3.5 w-3.5" />
   452→            {compareMode ? 'Exit Compare' : 'Compare'}
   453→          </Button>
   454→          {compareMode && selectedIds.size > 0 && (
   455→            <>
   456→              <span className="text-xs text-viralyze-muted">
   457→                {selectedIds.size}/3 selected
   458→              </span>
   459→              {selectedIds.size >= 2 && (
   460→                <Button
   461→                  size="sm"
   462→                  onClick={handleCompare}
   463→                  className="bg-wine-accent hover:bg-wine-accent/80 text-white gap-1.5"
   464→                >
   465→                  Compare Selected ({selectedIds.size})
   466→                </Button>
   467→              )}
   468→              <Button
   469→                size="sm"
   470→                variant="ghost"
   471→                onClick={() => setSelectedIds(new Set())}
   472→                className="text-viralyze-muted hover:text-viralyze-white gap-1"
   473→              >
   474→                <X className="h-3 w-3" />
   475→                Clear
   476→              </Button>
   477→            </>
   478→          )}
   479→        </motion.div>
   480→      )}
   481→
   482→      {/* Loading skeleton */}
   483→      {loading && (
   484→        <motion.div variants={item} className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   485→          {Array.from({ length: 4 }).map((_, i) => (
   486→            <SkeletonCard key={i} />
   487→          ))}
   488→        </motion.div>
   489→      )}
   490→
   491→      {/* Empty state — no analyses at all — animated dashed border */}
   492→      {!loading && isEmpty && (
   493→        <motion.div variants={item}>
   494→          <div className="relative rounded-xl p-px animate-pulse-glow">
   495→            <div className="absolute inset-0 rounded-xl border-2 border-dashed border-white/[0.08]" />
   496→            <Card className="glass relative z-0">
   497→              <CardContent className="p-12 flex flex-col items-center gap-3 text-center">
   498→                <motion.div
   499→                  animate={{ y: [0, -6, 0] }}
   500→                  transition={{ duration: 2, repeat: Infinity, ease: 'easeInOut' }}
   501→                >
   502→                  <Inbox className="h-10 w-10 text-viralyze-muted/40" />
   503→                </motion.div>
   504→                <p className="text-viralyze-muted text-sm">No analyses yet</p>
   505→                <p className="text-viralyze-muted/60 text-xs">Your saved content predictions will appear here</p>
   506→                <Button
   507→                  onClick={() => useAppStore.getState().setCurrentView('predict')}
   508→                  className="bg-gradient-wine hover:opacity-90 text-white mt-2 btn-shine"
   509→                >
   510→                  Analyze Content
   511→                </Button>
   512→              </CardContent>
   513→            </Card>
   514→          </div>
   515→        </motion.div>
   516→      )}
   517→
   518→      {/* Empty state — filtered results — animated dashed border */}
   519→      {isFilteredEmpty && (
   520→        <motion.div variants={item}>
   521→          <div className="relative rounded-xl p-px">
   522→            <div className="absolute inset-0 rounded-xl border-2 border-dashed border-white/[0.08]" />
   523→            <Card className="glass relative z-0">
   524→              <CardContent className="p-12 flex flex-col items-center gap-3 text-center">
   525→                <Search className="h-10 w-10 text-viralyze-muted/40" />
   526→                <p className="text-viralyze-muted text-sm">No results found</p>
   527→                <p className="text-viralyze-muted/60 text-xs">Try adjusting your search or filter</p>
   528→                <Button
   529→                  variant="outline"
   530→                  onClick={() => { setSearch(''); setPlatformFilter('all'); }}
   531→                  className="mt-2 border-white/10 text-viralyze-white hover:bg-white/[0.05] hover:border-white/20 transition-colors"
   532→                >
   533→                  Clear Filters
   534→                </Button>
   535→              </CardContent>
   536→            </Card>
   537→          </div>
   538→        </motion.div>
   539→      )}
   540→
   541→      {/* Results */}
   542→      {!loading && filtered.length > 0 && (
   543→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   544→          {filtered.map((analysis) => {
   545→            const PIcon = platformIcons[analysis.platform];
   546→            const isSelected = selectedIds.has(analysis.id);
   547→
   548→            // Compute top 3 score categories
