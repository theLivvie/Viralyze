   630→          <div className="relative rounded-xl p-px animate-pulse-glow">
   631→            <div className="absolute inset-0 rounded-xl border-2 border-dashed border-white/[0.08]" />

   645→                >
   646→                  Analyze Content
   647→                </Button>
   648→              </CardContent>
   649→            </Card>
   650→          </div>
   651→        </motion.div>
   652→      )}
   653→
   654→      {/* Empty state — filtered results — animated dashed border */}
   655→      {isFilteredEmpty && (
   656→        <motion.div variants={item}>
   657→          <div className="relative rounded-xl p-px">
   658→            <div className="absolute inset-0 rounded-xl border-2 border-dashed border-white/[0.08]" />

   734→                    {topScores && (
   735→                      <div className="flex items-center gap-2 mb-3">
   736→                        {topScores.map(({ key, value }) => (

   738→                            key={key}
   739→                            className="flex flex-col items-center gap-0.5 min-w-0"
   740→                          >
   741→                            <span className="text-[10px] leading-tight text-viralyze-muted/70 font-medium tabular-nums whitespace-nowrap">

   744→                            <div className="h-[1.5px] w-10 rounded-full bg-white/[0.06] overflow-hidden">
   745→                              <div
   746→                                className={cn('h-full rounded-full', scoreBarColor(value))}
   747→                                style={{ width: `${value}%` }}
   748→                              />
   749→                            </div>
   750→                          </div>
   751→                        ))}
   752→                      </div>
   753→                    )}
   754→                    <div className="flex items-center justify-between">
   755→                      <div className="flex items-center gap-2 text-xs text-viralyze-muted">
   756→                        <Badge variant="outline" className="text-xs border-white/10 text-viralyze-muted">
   757→                          {analysis.contentType}
   758→                        </Badge>
   759→                        <span>{new Date(analysis.createdAt).toLocaleDateString()}</span>
   760→                      </div>
   761→                      <Button
   762→                        variant="ghost"
   763→                        size="icon"
   764→                        className="h-7 w-7 text-viralyze-muted/50 hover:text-red-400 opacity-0 group-hover:opacity-100 transition-opacity"
   765→                        onClick={(e) => handleDelete(e, analysis.id)}
   766→                        disabled={deleting === analysis.id}
   767→                      >
   768→                        {deleting === analysis.id ? (
   769→                          <Loader2 className="h-3.5 w-3.5 animate-spin" />
   770→                        ) : (
   771→                          <Trash2 className="h-3.5 w-3.5" />
   772→                        )}
   773→                      </Button>
   774→                    </div>
   775→                  </CardContent>
   776→                </Card>
   777→              </motion.div>
   778→            );
   779→          })}
   780→        </div>
   781→      )}
   782→
   783→      {/* Compare Modal */}
   784→      <CompareModal
   785→        analyses={selectedAnalyses}
   786→        open={compareOpen}
   787→        onOpenChange={setCompareOpen}
   788→      />
   789→    </motion.div>
   790→  );
