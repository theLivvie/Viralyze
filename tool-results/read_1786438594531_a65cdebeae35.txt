   112→function CompareModal({
   113→  analyses: selected,
   114→  open,
   115→  onOpenChange,
   116→}: {
   117→  analyses: SavedAnalysis[];
   118→  open: boolean;
   119→  onOpenChange: (open: boolean) => void;
   120→}) {
   121→  const scoreColor = (score: number) => {
   122→    if (score >= 85) return 'text-emerald-400';
   123→    if (score >= 65) return 'text-green-400';
   124→    if (score >= 40) return 'text-amber-400';
   125→    return 'text-red-400';
   126→  };
   127→
   128→  const bestCategoryLabels: Record<string, string> = {
   129→    hook: 'Hook',
   130→    engagement: 'Engagement',
   131→    shareability: 'Shareability',
   132→    retention: 'Retention',
   133→    originality: 'Originality',
   134→    audienceFit: 'Audience Fit',
   135→    emotionalImpact: 'Emotional',
   136→    contentQuality: 'Quality',
   137→    trendAlignment: 'Trend',
   138→  };
   139→
   140→  // Compute best category per analysis
   141→  const getBestCategory = (a: SavedAnalysis) => {
   142→    if (!a.scores) return null;
   143→    const entries = Object.entries(a.scores).filter(([, v]) => typeof v === 'number');
   144→    if (entries.length === 0) return null;
   145→    entries.sort(([, a], [, b]) => (b as number) - (a as number));
   146→    const [key, value] = entries[0];
   147→    return { label: bestCategoryLabels[key] || key, value: value as number };
   148→  };
   149→
   150→  // Find winner IDs
   151→  const scoreWinnerId = selected.reduce<string | null>((
   152→    best,
   153→    a,
   154→  ) => {
   155→    const bestScore = selected.find((s) => s.id === best)?.overallScore ?? -1;
   156→    return a.overallScore > bestScore ? a.id : best;
   157→  }, selected[0]?.id ?? null);
   158→
   159→  const bestCategories = selected.map((a) => ({ id: a.id, best: getBestCategory(a) }));
   160→  const bestCategoryWinnerId = bestCategories.reduce<string | null>((
   161→    best,
   162→    curr,
   163→  ) => {
   164→    if (!curr.best) return best;
   165→    const bestVal = bestCategories.find((b) => b.id === best)?.best?.value ?? -1;
   166→    return curr.best.value > bestVal ? curr.id : best;
   167→  }, bestCategories[0]?.id ?? null);
   168→
   169→  return (
   170→    <Dialog open={open} onOpenChange={onOpenChange}>
   171→      <DialogContent className="bg-viralyze-black border-white/[0.08] max-w-3xl max-h-[80vh] overflow-y-auto">
   172→        <DialogHeader>
   173→          <DialogTitle className="text-viralyze-white flex items-center gap-2">
   174→            <GitCompareArrows className="h-5 w-5 text-wine-accent" />
   175→            Compare Analyses
   176→          </DialogTitle>
   177→          <DialogDescription className="text-viralyze-muted">
   178→            Side-by-side comparison of {selected.length} content analyses
   179→          </DialogDescription>
   180→        </DialogHeader>
   181→        <div className="mt-4">
   182→          <div className="overflow-x-auto">
   183→            <table className="w-full text-sm">
   184→              <thead>
   185→                <tr className="border-b border-white/[0.08]">
   186→                  <th className="text-left py-2.5 pr-4 text-viralyze-muted font-medium text-xs uppercase tracking-wider w-28">Field</th>
   187→                  {selected.map((a) => (
   188→                    <th key={a.id} className="text-left py-2.5 px-3 text-viralyze-white font-medium text-xs min-w-[180px]">
   189→                      {a.title || 'Untitled'}
   190→                    </th>
   191→                  ))}
   192→                </tr>
   193→              </thead>
   194→              <tbody className="divide-y divide-white/[0.04]">
   195→                {/* Score */}
   196→                <tr className="hover:glow-wine-sm transition-all">
   197→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Score</td>
   198→                  {selected.map((a) => (
   199→                    <td key={a.id} className="py-3 px-3">
   200→                      <span className={cn('text-lg font-bold tabular-nums', scoreColor(a.overallScore))}>
   201→                        {a.overallScore}
   202→                      </span>
   203→                      {scoreWinnerId === a.id && <span className="ml-1.5">🏆</span>}
   204→                    </td>
   205→                  ))}
   206→                </tr>
   207→                {/* Classification */}
   208→                <tr className="hover:glow-wine-sm transition-all">
   209→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Classification</td>
   210→                  {selected.map((a) => (
   211→                    <td key={a.id} className="py-3 px-3">
   212→                      <Badge
   213→                        variant="outline"
   214→                        className={cn('text-xs capitalize', classificationStyles[a.classification])}
   215→                      >
   216→                        {a.classification}
   217→                      </Badge>
   218→                    </td>
   219→                  ))}
   220→                </tr>
   221→                {/* Platform */}
   222→                <tr className="hover:glow-wine-sm transition-all">
   223→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Platform</td>
   224→                  {selected.map((a) => {
   225→                    const PI = platformIcons[a.platform];
   226→                    return (
   227→                      <td key={a.id} className="py-3 px-3">
   228→                        <div className="flex items-center gap-1.5 text-viralyze-white">
   229→                          <PI className="h-3.5 w-3.5 text-viralyze-muted" />
   230→                          <span className="text-xs capitalize">{a.platform}</span>
   231→                        </div>
   232→                      </td>
   233→                    );
   234→                  })}
   235→                </tr>
   236→                {/* Content Type */}
   237→                <tr className="hover:glow-wine-sm transition-all">
   238→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Content Type</td>
   239→                  {selected.map((a) => (
   240→                    <td key={a.id} className="py-3 px-3">
   241→                      <span className="text-xs text-viralyze-white capitalize">{a.contentType}</span>
   242→                    </td>
   243→                  ))}
   244→                </tr>
   245→                {/* Date */}
   246→                <tr className="hover:glow-wine-sm transition-all">
   247→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Date</td>
   248→                  {selected.map((a) => (
   249→                    <td key={a.id} className="py-3 px-3">
   250→                      <span className="text-xs text-viralyze-muted">
   251→                        {new Date(a.createdAt).toLocaleDateString('en-US', {
   252→                          month: 'short',
   253→                          day: 'numeric',
   254→                          year: 'numeric',
   255→                        })}
   256→                      </span>
   257→                    </td>
   258→                  ))}
   259→                </tr>
   260→                {/* Best Category */}
   261→                <tr className="hover:glow-wine-sm transition-all">
   262→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Best Category</td>
   263→                  {bestCategories.map((bc) => (
   264→                    <td key={bc.id} className="py-3 px-3">
   265→                      {bc.best ? (
   266→                        <div className="flex items-center gap-1.5">
   267→                          <span className="text-xs text-viralyze-white font-medium">{bc.best.label}</span>
   268→                          <span className={cn('text-xs font-bold tabular-nums', scoreColor(bc.best.value))}>
   269→                            {bc.best.value}
   270→                          </span>
   271→                          {bestCategoryWinnerId === bc.id && <span>🏆</span>}
   272→                        </div>
   273→                      ) : (
   274→                        <span className="text-xs text-viralyze-muted">N/A</span>
   275→                      )}
   276→                    </td>
   277→                  ))}
   278→                </tr>
   279→                {/* Visual score bar */}
   280→                <tr className="hover:glow-wine-sm transition-all">
   281→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Score Bar</td>
   282→                  {selected.map((a) => {
   283→                    const barColor = a.overallScore >= 70 ? 'bg-green-400' : a.overallScore >= 50 ? 'bg-amber-400' : 'bg-red-400';
   284→                    return (
   285→                      <td key={a.id} className="py-3 px-3">
   286→                        <div className="h-2.5 w-full max-w-[140px] rounded-full bg-white/[0.06] overflow-hidden">
   287→                          <div
   288→                            className={cn('h-full rounded-full transition-all duration-500', barColor)}
   289→                            style={{ width: `${a.overallScore}%` }}
   290→                          />
   291→                        </div>
   292→                      </td>
   293→                    );
   294→                  })}
   295→                </tr>
   296→              </tbody>
   297→            </table>
   298→          </div>
   299→        </div>
   300→      </DialogContent>
   301→    </Dialog>
   302→  );
   303→}

   305→export default function LibraryView() {
   306→  const { setSavedAnalyses, removeSavedAnalysis, setCurrentAnalysis, setCurrentView, user } = useAppStore();
   307→  const [analyses, setAnalyses] = useState<SavedAnalysis[]>([]);
   308→  const [loading, setLoading] = useState(true);
   309→  const [search, setSearch] = useState('');
   310→  const [platformFilter, setPlatformFilter] = useState<string>('all');
   311→  const [sort, setSort] = useState<string>('newest');
   312→  const [deleting, setDeleting] = useState<string | null>(null);
   313→  const [compareMode, setCompareMode] = useState(false);
   314→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   315→  const [compareOpen, setCompareOpen] = useState(false);
   316→  const [confirmDeleteId, setConfirmDeleteId] = useState<string | null>(null);
   317→  const [noteExpandedId, setNoteExpandedId] = useState<string | null>(null);
   318→  const [noteDrafts, setNoteDrafts] = useState<Record<string, string>>({});
   319→  const [notesFilter, setNotesFilter] = useState(false);
   320→
   321→  const fetchLibrary = useCallback(async () => {
   322→    if (!user?.id) {
   323→      setLoading(false);
   324→      return;
   325→    }
   326→    try {
   327→      setLoading(true);
   328→      const res = await fetch(`/api/library?userId=${user.id}`);
   329→      if (res.ok) {
   330→        const data = await res.json();
   331→        setAnalyses(data);
   332→        setSavedAnalyses(data);
   333→      }
   334→    } catch {
   335→      // Silently fail — use whatever is in local state
   336→    } finally {
   337→      setLoading(false);
   338→    }
   339→  }, [user?.id, setSavedAnalyses]);
   340→
   341→  useEffect(() => {
   342→    fetchLibrary();
   343→  }, [fetchLibrary]);
   344→
   345→  // Notes helpers
   346→  const getNote = (id: string): string => {
   347→    if (noteDrafts[id] !== undefined) return noteDrafts[id];
   348→    try {
   349→      return localStorage.getItem(`viralyze-notes-${id}`) || '';
   350→    } catch {
   351→      return '';
   352→    }
   353→  };
   354→
   355→  const hasNote = (id: string): boolean => {
   356→    if (noteDrafts[id] !== undefined) return noteDrafts[id].length > 0;
   357→    try {
   358→      return (localStorage.getItem(`viralyze-notes-${id}`) || '').length > 0;
   359→    } catch {
   360→      return false;
   361→    }
   362→  };
   363→
   364→  const saveNote = (id: string, text: string) => {
   365→    setNoteDrafts((prev) => ({ ...prev, [id]: text }));
   366→    try {
   367→      if (text.trim()) {
   368→        localStorage.setItem(`viralyze-notes-${id}`, text);
   369→      } else {
   370→        localStorage.removeItem(`viralyze-notes-${id}`);
   371→      }
   372→    } catch { /* noop */ }
   373→  };
   374→
   375→  const toggleNote = (e: React.MouseEvent, id: string) => {
   376→    e.stopPropagation();
   377→    setNoteExpandedId((prev) => (prev === id ? null : id));
   378→  };
   379→
   380→  const filtered = analyses
   381→    .filter((a) => {
   382→      if (notesFilter && !hasNote(a.id)) return false;
   383→      if (search && !a.title.toLowerCase().includes(search.toLowerCase())) return false;
   384→      if (platformFilter !== 'all' && a.platform !== platformFilter) return false;
   385→      return true;
   386→    })
   387→    .sort((a, b) => {
   388→      if (sort === 'newest') return new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
   389→      if (sort === 'oldest') return new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime();
   390→      if (sort === 'score-high') return b.overallScore - a.overallScore;
   391→      if (sort === 'score-low') return a.overallScore - b.overallScore;
   392→      return 0;
   393→    });
   394→
   395→  const handleDelete = async (e: React.MouseEvent, id: string) => {
   396→    e.stopPropagation();
   397→    if (!user?.id) return;
   398→    if (confirmDeleteId !== id) {
   399→      setConfirmDeleteId(id);
   400→      setTimeout(() => setConfirmDeleteId(null), 2000);
   401→      return;
   402→    }
   403→    setDeleting(id);
   404→    try {
   405→      const res = await fetch(`/api/library?id=${id}&userId=${user.id}`, { method: 'DELETE' });
   406→      if (res.ok) {
   407→        setAnalyses((prev) => prev.filter((a) => a.id !== id));
   408→        removeSavedAnalysis(id);
   409→        setSelectedIds((prev) => { const next = new Set(prev); next.delete(id); return next; });
   410→        toast.success('Removed from library');
   411→      } else {
   412→        toast.error('Failed to delete analysis');
   413→      }
   414→    } catch {
   415→      toast.error('Failed to delete analysis');
   416→    } finally {
   417→      setDeleting(null);
   418→    }
   419→  };
   420→
   421→  const [viewingId, setViewingId] = useState<string | null>(null);
   422→
   423→  const handleCardClick = async (id: string) => {
   424→    if (compareMode) {
   425→      setSelectedIds((prev) => {
   426→        const next = new Set(prev);
   427→        if (next.has(id)) {
   428→          next.delete(id);
   429→        } else if (next.size < 3) {
   430→          next.add(id);
   431→        } else {
   432→          toast.info('You can compare up to 3 items');
   433→          return prev;
   434→        }
   435→        return next;
   436→      });
   437→    } else {
   438→      try {
   439→        setViewingId(id);
   440→        const res = await fetch(`/api/library?id=${id}`);
   441→        if (res.ok) {
   442→          const data = await res.json();
   443→          setCurrentAnalysis(data);
   444→          setCurrentView('analysis');
   445→        } else {
   446→          toast.error('Failed to load analysis');
   447→        }
   448→      } catch {
   449→        toast.error('Failed to load analysis');
   450→      } finally {
   451→        setViewingId(null);
   452→      }
   453→    }
   454→  };
   455→
   456→  const selectedAnalyses = analyses.filter((a) => selectedIds.has(a.id));
   457→
   458→  const handleCompare = () => {
   459→    if (selectedIds.size >= 2) {
   460→      setCompareOpen(true);
   461→    }
   462→  };
   463→
   464→  const exitCompareMode = () => {
   465→    setCompareMode(false);
   466→    setSelectedIds(new Set());
   467→  };
   468→
   469→  const isFilteredEmpty = !loading && filtered.length === 0 && analyses.length > 0;
   470→  const isEmpty = !loading && analyses.length === 0;
   471→
   472→  const compareItems = compareMode && selectedIds.size === 2
   473→    ? analyses.filter((a) => selectedIds.has(a.id))
   474→    : [];
   475→  const showComparePanel = compareItems.length === 2;
   476→  const [compareA, compareB] = showComparePanel ? compareItems : [null, null];
   477→  const compareScoreDiff = showComparePanel ? compareA!.overallScore - compareB!.overallScore : 0;
   478→
   479→  return (
   480→    <motion.div
   481→      variants={container}
   482→      initial="hidden"
   483→      animate="show"
   484→      className="flex flex-col gap-4 max-w-4xl mx-auto"
   485→    >
   486→      {/* Header with count badge */}
   487→      <motion.div variants={item} className="flex items-center gap-3">
   488→        <div className="flex flex-col">
   489→          <div className="flex items-center gap-2.5">
   490→            <h2 className="text-2xl md:text-3xl font-bold text-viralyze-white">
   491→              Content Library
   492→            </h2>
   493→            {!loading && analyses.length > 0 && (
   494→              <Badge
   495→                variant="outline"
   496→                className="text-xs border-wine-accent/30 text-wine-accent bg-wine-accent/10 tabular-nums"
   497→              >
   498→                {analyses.length}
   499→              </Badge>
   500→            )}
   501→          </div>
   502→          <p className="text-viralyze-muted mt-1">
   503→            Your saved content analyses
   504→          </p>
   505→        </div>
   506→      </motion.div>
   507→
   508→      {/* Filters */}
   509→      <motion.div variants={item} className="flex flex-col sm:flex-row gap-3">
   510→        <div className="relative flex-1">
   511→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-viralyze-muted" />
   512→          <Input
   513→            placeholder="Search analyses..."
   514→            value={search}
   515→            onChange={(e) => setSearch(e.target.value)}
   516→            className="pl-9 bg-white/[0.05] border-white/[0.08] text-viralyze-white placeholder:text-viralyze-muted/40 focus-visible:ring-wine-accent hover:border-white/[0.12] transition-colors"
   517→          />
   518→        </div>
   519→        <Select value={platformFilter} onValueChange={setPlatformFilter}>
   520→          <SelectTrigger className="w-full sm:w-40 bg-white/[0.05] border-white/[0.08] text-viralyze-white hover:border-white/[0.12] transition-colors">
   521→            <SelectValue placeholder="Platform" />
   522→          </SelectTrigger>
   523→          <SelectContent className="bg-viralyze-soft-black border-white/[0.08]">
   524→            <SelectItem value="all" className="text-viralyze-white">All Platforms</SelectItem>
   525→            <SelectItem value="instagram" className="text-viralyze-white">Instagram</SelectItem>
   526→            <SelectItem value="youtube" className="text-viralyze-white">YouTube</SelectItem>
   527→            <SelectItem value="tiktok" className="text-viralyze-white">TikTok</SelectItem>
   528→            <SelectItem value="x" className="text-viralyze-white">X</SelectItem>
   529→            <SelectItem value="linkedin" className="text-viralyze-white">LinkedIn</SelectItem>
   530→          </SelectContent>
   531→        </Select>
   532→        <Select value={sort} onValueChange={setSort}>
   533→          <SelectTrigger className="w-full sm:w-40 bg-white/[0.05] border-white/[0.08] text-viralyze-white hover:border-white/[0.12] transition-colors">
   534→            <SelectValue placeholder="Sort" />
   535→          </SelectTrigger>
   536→          <SelectContent className="bg-viralyze-soft-black border-white/[0.08]">
