   125→   125→// --- Animated Counter Component ---
   126→   126→function AnimatedCounter({ value, duration = 1.2 }: { value: number; duration?: number }) {
   127→   127→  const [display, setDisplay] = useState(0);
   128→   128→  const ref = useRef(false);
   129→   129→  const mv = useMotionValue(0);
   130→   130→  const displayValue = useTransform(mv, (v) => Math.round(v));
   131→   131→
   132→   132→  useEffect(() => {
   133→   133→    const unsub = displayValue.on('change', (v) => setDisplay(v));
   134→   134→    return unsub;
   135→   135→  }, [displayValue]);
   136→   136→
   137→   137→  useEffect(() => {
   138→   138→    if (ref.current) return;
   139→   139→    ref.current = true;
   140→   140→    const ctrl = animate(mv, value, { duration, ease: 'easeOut' });
   141→   141→    return () => { ctrl.stop(); };
   142→   142→  }, [value, duration]);
   143→   143→
   144→   144→  return <>{display}</>;
   145→   145→}
