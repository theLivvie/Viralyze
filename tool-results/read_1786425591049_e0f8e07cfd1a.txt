   111→function CompareModal({
   112→  analyses: selected,
   113→  open,
   114→  onOpenChange,
   115→}: {
   116→  analyses: SavedAnalysis[];
   117→  open: boolean;
   118→  onOpenChange: (open: boolean) => void;
   119→}) {
   120→  const scoreColor = (score: number) => {
   121→    if (score >= 85) return 'text-emerald-400';
   122→    if (score >= 65) return 'text-green-400';
   123→    if (score >= 40) return 'text-amber-400';
   124→    return 'text-red-400';
   125→  };
   126→
   127→  const bestCategoryLabels: Record<string, string> = {
   128→    hook: 'Hook',
   129→    engagement: 'Engagement',
   130→    shareability: 'Shareability',
   131→    retention: 'Retention',
   132→    originality: 'Originality',
   133→    audienceFit: 'Audience Fit',
   134→    emotionalImpact: 'Emotional',
   135→    contentQuality: 'Quality',
   136→    trendAlignment: 'Trend',
   137→  };
   138→
   139→  // Compute best category per analysis
   140→  const getBestCategory = (a: SavedAnalysis) => {
   141→    if (!a.scores) return null;
   142→    const entries = Object.entries(a.scores).filter(([, v]) => typeof v === 'number');
   143→    if (entries.length === 0) return null;
   144→    entries.sort(([, a], [, b]) => (b as number) - (a as number));
   145→    const [key, value] = entries[0];
   146→    return { label: bestCategoryLabels[key] || key, value: value as number };
   147→  };
   148→
   149→  // Find winner IDs
   150→  const scoreWinnerId = selected.reduce<string | null>((
   151→    best,
   152→    a,
   153→  ) => {
   154→    const bestScore = selected.find((s) => s.id === best)?.overallScore ?? -1;
   155→    return a.overallScore > bestScore ? a.id : best;
   156→  }, selected[0]?.id ?? null);
   157→
   158→  const bestCategories = selected.map((a) => ({ id: a.id, best: getBestCategory(a) }));
   159→  const bestCategoryWinnerId = bestCategories.reduce<string | null>((
   160→    best,
   161→    curr,
   162→  ) => {
   163→    if (!curr.best) return best;
   164→    const bestVal = bestCategories.find((b) => b.id === best)?.best?.value ?? -1;
   165→    return curr.best.value > bestVal ? curr.id : best;
   166→  }, bestCategories[0]?.id ?? null);
   167→
   168→  return (
   169→    <Dialog open={open} onOpenChange={onOpenChange}>
   170→      <DialogContent className="bg-viralyze-black border-white/[0.08] max-w-3xl max-h-[80vh] overflow-y-auto">
   171→        <DialogHeader>
   172→          <DialogTitle className="text-viralyze-white flex items-center gap-2">
   173→            <GitCompareArrows className="h-5 w-5 text-wine-accent" />
   174→            Compare Analyses
   175→          </DialogTitle>
   176→          <DialogDescription className="text-viralyze-muted">
   177→            Side-by-side comparison of {selected.length} content analyses
   178→          </DialogDescription>
   179→        </DialogHeader>
   180→        <div className="mt-4">
   181→          <div className="overflow-x-auto">
   182→            <table className="w-full text-sm">
   183→              <thead>
   184→                <tr className="border-b border-white/[0.08]">
   185→                  <th className="text-left py-2.5 pr-4 text-viralyze-muted font-medium text-xs uppercase tracking-wider w-28">Field</th>
   186→                  {selected.map((a) => (
   187→                    <th key={a.id} className="text-left py-2.5 px-3 text-viralyze-white font-medium text-xs min-w-[180px]">
   188→                      {a.title || 'Untitled'}
   189→                    </th>
   190→                  ))}
   191→                </tr>
   192→              </thead>
   193→              <tbody className="divide-y divide-white/[0.04]">
   194→                {/* Score */}
   195→                <tr className="hover:glow-wine-sm transition-all">
   196→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Score</td>
   197→                  {selected.map((a) => (
   198→                    <td key={a.id} className="py-3 px-3">
   199→                      <span className={cn('text-lg font-bold tabular-nums', scoreColor(a.overallScore))}>
   200→                        {a.overallScore}
   201→                      </span>
   202→                      {scoreWinnerId === a.id && <span className="ml-1.5">🏆</span>}
   203→                    </td>
   204→                  ))}
   205→                </tr>
   206→                {/* Classification */}
   207→                <tr className="hover:glow-wine-sm transition-all">
   208→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Classification</td>
   209→                  {selected.map((a) => (
   210→                    <td key={a.id} className="py-3 px-3">
   211→                      <Badge
   212→                        variant="outline"
   213→                        className={cn('text-xs capitalize', classificationStyles[a.classification])}
   214→                      >
   215→                        {a.classification}
   216→                      </Badge>
   217→                    </td>
   218→                  ))}
   219→                </tr>
   220→                {/* Platform */}
   221→                <tr className="hover:glow-wine-sm transition-all">
   222→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Platform</td>
   223→                  {selected.map((a) => {
   224→                    const PI = platformIcons[a.platform];
   225→                    return (
   226→                      <td key={a.id} className="py-3 px-3">
   227→                        <div className="flex items-center gap-1.5 text-viralyze-white">
   228→                          <PI className="h-3.5 w-3.5 text-viralyze-muted" />
   229→                          <span className="text-xs capitalize">{a.platform}</span>
   230→                        </div>
   231→                      </td>
   232→                    );
   233→                  })}
   234→                </tr>
   235→                {/* Content Type */}
   236→                <tr className="hover:glow-wine-sm transition-all">
   237→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Content Type</td>
   238→                  {selected.map((a) => (
   239→                    <td key={a.id} className="py-3 px-3">
   240→                      <span className="text-xs text-viralyze-white capitalize">{a.contentType}</span>
   241→                    </td>
   242→                  ))}
   243→                </tr>
   244→                {/* Date */}
   245→                <tr className="hover:glow-wine-sm transition-all">
   246→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Date</td>
   247→                  {selected.map((a) => (
   248→                    <td key={a.id} className="py-3 px-3">
   249→                      <span className="text-xs text-viralyze-muted">
   250→                        {new Date(a.createdAt).toLocaleDateString('en-US', {
   251→                          month: 'short',
   252→                          day: 'numeric',
   253→                          year: 'numeric',
   254→                        })}
   255→                      </span>
   256→                    </td>
   257→                  ))}
   258→                </tr>
   259→                {/* Best Category */}
   260→                <tr className="hover:glow-wine-sm transition-all">
   261→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Best Category</td>
   262→                  {bestCategories.map((bc) => (
   263→                    <td key={bc.id} className="py-3 px-3">
   264→                      {bc.best ? (
   265→                        <div className="flex items-center gap-1.5">
   266→                          <span className="text-xs text-viralyze-white font-medium">{bc.best.label}</span>
   267→                          <span className={cn('text-xs font-bold tabular-nums', scoreColor(bc.best.value))}>
   268→                            {bc.best.value}
   269→                          </span>
   270→                          {bestCategoryWinnerId === bc.id && <span>🏆</span>}
   271→                        </div>
   272→                      ) : (
   273→                        <span className="text-xs text-viralyze-muted">N/A</span>
   274→                      )}
   275→                    </td>
   276→                  ))}
   277→                </tr>
   278→                {/* Visual score bar */}
   279→                <tr className="hover:glow-wine-sm transition-all">
   280→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Score Bar</td>
   281→                  {selected.map((a) => {
   282→                    const barColor = a.overallScore >= 70 ? 'bg-green-400' : a.overallScore >= 50 ? 'bg-amber-400' : 'bg-red-400';
   283→                    return (
   284→                      <td key={a.id} className="py-3 px-3">
   285→                        <div className="h-2.5 w-full max-w-[140px] rounded-full bg-white/[0.06] overflow-hidden">
   286→                          <div
   287→                            className={cn('h-full rounded-full transition-all duration-500', barColor)}
   288→                            style={{ width: `${a.overallScore}%` }}
   289→                          />
   290→                        </div>
   291→                      </td>
   292→                    );
   293→                  })}
   294→                </tr>
   295→              </tbody>
   296→            </table>
   297→          </div>
   298→        </div>
   299→      </DialogContent>
   300→    </Dialog>
   301→  );
   302→}

   304→export default function LibraryView() {
   305→  const { setSavedAnalyses, removeSavedAnalysis, setCurrentAnalysis, setCurrentView, user } = useAppStore();
   306→  const [analyses, setAnalyses] = useState<SavedAnalysis[]>([]);
   307→  const [loading, setLoading] = useState(true);
   308→  const [search, setSearch] = useState('');
   309→  const [platformFilter, setPlatformFilter] = useState<string>('all');
   310→  const [sort, setSort] = useState<string>('newest');
   311→  const [deleting, setDeleting] = useState<string | null>(null);
   312→  const [compareMode, setCompareMode] = useState(false);
   313→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   314→  const [compareOpen, setCompareOpen] = useState(false);
   315→  const [confirmDeleteId, setConfirmDeleteId] = useState<string | null>(null);
   316→
   317→  const fetchLibrary = useCallback(async () => {
   318→    if (!user?.id) {
   319→      setLoading(false);
   320→      return;
   321→    }
   322→    try {
   323→      setLoading(true);
   324→      const res = await fetch(`/api/library?userId=${user.id}`);
   325→      if (res.ok) {
   326→        const data = await res.json();
   327→        setAnalyses(data);
   328→        setSavedAnalyses(data);
   329→      }
   330→    } catch {
   331→      // Silently fail — use whatever is in local state
   332→    } finally {
   333→      setLoading(false);
   334→    }
   335→  }, [user?.id, setSavedAnalyses]);
   336→
   337→  useEffect(() => {
   338→    fetchLibrary();
   339→  }, [fetchLibrary]);
   340→
   341→  const filtered = analyses
   342→    .filter((a) => {
   343→      if (search && !a.title.toLowerCase().includes(search.toLowerCase())) return false;
   344→      if (platformFilter !== 'all' && a.platform !== platformFilter) return false;
   345→      return true;
   346→    })
   347→    .sort((a, b) => {
   348→      if (sort === 'newest') return new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
   349→      if (sort === 'oldest') return new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime();
   350→      if (sort === 'score-high') return b.overallScore - a.overallScore;
   351→      if (sort === 'score-low') return a.overallScore - b.overallScore;
   352→      return 0;
   353→    });
   354→
   355→  const handleDelete = async (e: React.MouseEvent, id: string) => {
   356→    e.stopPropagation();
   357→    if (!user?.id) return;
   358→    if (confirmDeleteId !== id) {
   359→      setConfirmDeleteId(id);
   360→      setTimeout(() => setConfirmDeleteId(null), 2000);
   361→      return;
   362→    }
   363→    setDeleting(id);
   364→    try {
   365→      const res = await fetch(`/api/library?id=${id}&userId=${user.id}`, { method: 'DELETE' });
   366→      if (res.ok) {
   367→        setAnalyses((prev) => prev.filter((a) => a.id !== id));
   368→        removeSavedAnalysis(id);
   369→        setSelectedIds((prev) => { const next = new Set(prev); next.delete(id); return next; });
   370→        toast.success('Removed from library');
   371→      } else {
   372→        toast.error('Failed to delete analysis');
   373→      }
   374→    } catch {
   375→      toast.error('Failed to delete analysis');
   376→    } finally {
   377→      setDeleting(null);
   378→    }
   379→  };
   380→
   381→  const [viewingId, setViewingId] = useState<string | null>(null);
   382→
   383→  const handleCardClick = async (id: string) => {
   384→    if (compareMode) {
   385→      setSelectedIds((prev) => {
   386→        const next = new Set(prev);
   387→        if (next.has(id)) {
   388→          next.delete(id);
   389→        } else if (next.size < 3) {
   390→          next.add(id);
   391→        } else {
   392→          toast.info('You can compare up to 3 items');
   393→          return prev;
   394→        }
   395→        return next;
   396→      });
   397→    } else {
   398→      try {
   399→        setViewingId(id);
   400→        const res = await fetch(`/api/library?id=${id}`);
   401→        if (res.ok) {
   402→          const data = await res.json();
   403→          setCurrentAnalysis(data);
   404→          setCurrentView('analysis');
   405→        } else {
   406→          toast.error('Failed to load analysis');
   407→        }
   408→      } catch {
   409→        toast.error('Failed to load analysis');
   410→      } finally {
   411→        setViewingId(null);
   412→      }
   413→    }
   414→  };
   415→
   416→  const selectedAnalyses = analyses.filter((a) => selectedIds.has(a.id));
   417→
   418→  const handleCompare = () => {
   419→    if (selectedIds.size >= 2) {
   420→      setCompareOpen(true);
   421→    }
   422→  };
   423→
   424→  const exitCompareMode = () => {
   425→    setCompareMode(false);
   426→    setSelectedIds(new Set());
   427→  };
   428→
   429→  const isFilteredEmpty = !loading && filtered.length === 0 && analyses.length > 0;
   430→  const isEmpty = !loading && analyses.length === 0;
   431→
   432→  const compareItems = compareMode && selectedIds.size === 2
   433→    ? analyses.filter((a) => selectedIds.has(a.id))
   434→    : [];
   435→  const showComparePanel = compareItems.length === 2;
   436→  const [compareA, compareB] = showComparePanel ? compareItems : [null, null];
   437→  const compareScoreDiff = showComparePanel ? compareA!.overallScore - compareB!.overallScore : 0;
   438→
   439→  return (
   440→    <motion.div
   441→      variants={container}
   442→      initial="hidden"
   443→      animate="show"
   444→      className="flex flex-col gap-4 max-w-4xl mx-auto"
   445→    >
   446→      {/* Header with count badge */}
   447→      <motion.div variants={item} className="flex items-center gap-3">
   448→        <div className="flex flex-col">
   449→          <div className="flex items-center gap-2.5">
   450→            <h2 className="text-2xl md:text-3xl font-bold text-viralyze-white">
   451→              Content Library
   452→            </h2>
   453→            {!loading && analyses.length > 0 && (
   454→              <Badge
   455→                variant="outline"
   456→                className="text-xs border-wine-accent/30 text-wine-accent bg-wine-accent/10 tabular-nums"
   457→              >
   458→                {analyses.length}
   459→              </Badge>
   460→            )}
   461→          </div>
   462→          <p className="text-viralyze-muted mt-1">
   463→            Your saved content analyses
   464→          </p>
   465→        </div>
   466→      </motion.div>
   467→
   468→      {/* Filters */}
   469→      <motion.div variants={item} className="flex flex-col sm:flex-row gap-3">
   470→        <div className="relative flex-1">
   471→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-viralyze-muted" />
   472→          <Input
   473→            placeholder="Search analyses..."
   474→            value={search}
   475→            onChange={(e) => setSearch(e.target.value)}
   476→            className="pl-9 bg-white/[0.05] border-white/[0.08] text-viralyze-white placeholder:text-viralyze-muted/40 focus-visible:ring-wine-accent hover:border-white/[0.12] transition-colors"
   477→          />
   478→        </div>
   479→        <Select value={platformFilter} onValueChange={setPlatformFilter}>
   480→          <SelectTrigger className="w-full sm:w-40 bg-white/[0.05] border-white/[0.08] text-viralyze-white hover:border-white/[0.12] transition-colors">
   481→            <SelectValue placeholder="Platform" />
   482→          </SelectTrigger>
   483→          <SelectContent className="bg-viralyze-soft-black border-white/[0.08]">
   484→            <SelectItem value="all" className="text-viralyze-white">All Platforms</SelectItem>
   485→            <SelectItem value="instagram" className="text-viralyze-white">Instagram</SelectItem>
   486→            <SelectItem value="youtube" className="text-viralyze-white">YouTube</SelectItem>
   487→            <SelectItem value="tiktok" className="text-viralyze-white">TikTok</SelectItem>
   488→            <SelectItem value="x" className="text-viralyze-white">X</SelectItem>
   489→            <SelectItem value="linkedin" className="text-viralyze-white">LinkedIn</SelectItem>
   490→          </SelectContent>
   491→        </Select>
   492→        <Select value={sort} onValueChange={setSort}>
   493→          <SelectTrigger className="w-full sm:w-40 bg-white/[0.05] border-white/[0.08] text-viralyze-white hover:border-white/[0.12] transition-colors">
   494→            <SelectValue placeholder="Sort" />
   495→          </SelectTrigger>
   496→          <SelectContent className="bg-viralyze-soft-black border-white/[0.08]">
   497→            <SelectItem value="newest" className="text-viralyze-white">Newest First</SelectItem>
   498→            <SelectItem value="oldest" className="text-viralyze-white">Oldest First</SelectItem>
   499→            <SelectItem value="score-high" className="text-viralyze-white">Score: High to Low</SelectItem>
   500→            <SelectItem value="score-low" className="text-viralyze-white">Score: Low to High</SelectItem>
   501→          </SelectContent>
   502→        </Select>
   503→      </motion.div>
   504→
   505→      {/* Score Trend sparkline */}
   506→      {!loading && analyses.length >= 3 && (
   507→        <motion.div variants={item}>
   508→          <div className="flex items-center gap-2 mb-2">
   509→            <span className="text-xs font-medium text-viralyze-muted uppercase tracking-wider">Score Trend</span>
   510→          </div>
   511→          <Card className="glass">
   512→            <CardContent className="p-4">
   513→              {(() => {
   514→                const last15 = analyses.slice(-15);
   515→                const avg = Math.round(last15.reduce((s, a) => s + a.overallScore, 0) / last15.length);
   516→                return (
   517→                  <div className="relative flex items-end gap-1.5 h-20">
   518→                    {last15.map((a, i) => {
   519→                      const h = Math.max(4, (a.overallScore / 100) * 100);
   520→                      const barColor = a.overallScore >= 70 ? 'bg-green-400' : a.overallScore >= 50 ? 'bg-amber-400' : 'bg-red-400';
   521→                      return (
   522→                        <motion.div
   523→                          key={a.id}
   524→                          initial={{ height: 0 }}
   525→                          animate={{ height: `${h}%` }}
   526→                          transition={{ duration: 0.5, delay: i * 0.04, ease: 'easeOut' }}
   527→                          className={cn('flex-1 rounded-sm min-w-[4px] max-w-[20px]', barColor)}
