    65→    65→function getWeekDays(weekOffset: number): Date[] {
    66→    66→  const now = new Date();
    67→    67→  const currentDay = now.getDay();
    68→    68→  const mondayOffset = currentDay === 0 ? -6 : 1 - currentDay;
    69→    69→  const monday = new Date(now);
    70→    70→  monday.setDate(now.getDate() + mondayOffset + weekOffset * 7);
    71→    71→  monday.setHours(0, 0, 0, 0);
    72→    72→
    73→    73→  const days: Date[] = [];
    74→    74→  for (let i = 0; i < 7; i++) {
    75→    75→    const d = new Date(monday);
    76→    76→    d.setDate(monday.getDate() + i);
    77→    77→    days.push(d);
    78→    78→  }
    79→    79→  return days;
    80→    80→}

    86→    86→function formatDayLabel(d: Date): string {
    87→    87→  const today = new Date();
    88→    88→  today.setHours(0, 0, 0, 0);
    89→    89→  const target = new Date(d);
    90→    90→  target.setHours(0, 0, 0, 0);
    91→    91→  const diff = Math.round((target.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
    92→    92→  if (diff === 0) return 'Today';
    93→    93→  if (diff === 1) return 'Tomorrow';
    94→    94→  if (diff === -1) return 'Yesterday';
    95→    95→  return d.toLocaleDateString('en-US', { weekday: 'short', month: 'short', day: 'numeric' });
    96→    96→}
