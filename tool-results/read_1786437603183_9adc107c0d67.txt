    77→function getTimeOfDayIcon(timeStr: string) {
    78→  if (!timeStr) return Clock;
    79→  const hour = parseInt(timeStr.split(':')[0], 10);
    80→  if (hour >= 5 && hour < 12) return Sun;
    81→  if (hour >= 12 && hour < 18) return SunDim;
    82→  return Moon;
    83→}
    84→
    85→function getTimeOfDayColor(timeStr: string) {
    86→  if (!timeStr) return 'text-viralyze-muted';
    87→  const hour = parseInt(timeStr.split(':')[0], 10);
    88→  if (hour >= 5 && hour < 12) return 'text-amber-400';
    89→  if (hour >= 12 && hour < 18) return 'text-orange-400';
    90→  return 'text-purple-400';
    91→}

    93→function getWeekDays(weekOffset: number): Date[] {
    94→  const now = new Date();
    95→  const currentDay = now.getDay();
    96→  const mondayOffset = currentDay === 0 ? -6 : 1 - currentDay;
    97→  const monday = new Date(now);
    98→  monday.setDate(now.getDate() + mondayOffset + weekOffset * 7);
    99→  monday.setHours(0, 0, 0, 0);
   100→
   101→  const days: Date[] = [];
   102→  for (let i = 0; i < 7; i++) {
   103→    const d = new Date(monday);
   104→    d.setDate(monday.getDate() + i);
   105→    days.push(d);
   106→  }
   107→  return days;
   108→}

   110→function dateKey(d: Date): string {

   114→function formatDayLabel(d: Date): string {
   115→  const today = new Date();
   116→  today.setHours(0, 0, 0, 0);
   117→  const target = new Date(d);
   118→  target.setHours(0, 0, 0, 0);
   119→  const diff = Math.round((target.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
   120→  if (diff === 0) return 'Today';
   121→  if (diff === 1) return 'Tomorrow';
   122→  if (diff === -1) return 'Yesterday';
   123→  return d.toLocaleDateString('en-US', { weekday: 'short', month: 'short', day: 'numeric' });
   124→}

   136→function isToday(d: Date): boolean {
   137→  const today = new Date();
   138→  return d.getDate() === today.getDate() &&
   139→    d.getMonth() === today.getMonth() &&
   140→    d.getFullYear() === today.getFullYear();

   143→function loadSlots(): WeekSlots {
   144→  if (typeof window === 'undefined') return {};
   145→  try {
   146→    const raw = localStorage.getItem(STORAGE_KEY);
   147→    if (!raw) return {};
   148→    const parsed = JSON.parse(raw);
   149→    // Migrate old slots that may not have contentType
   150→    let needsSave = false;
   151→    for (const dk of Object.keys(parsed)) {
   152→      const updated = parsed[dk].map((slot: CalendarSlot) => {
   153→        if (!slot.contentType) { needsSave = true; return { ...slot, contentType: 'post' as ContentType }; }
   154→        return slot;
   155→      });
   156→      parsed[dk] = updated;
   157→    }
   158→    if (needsSave) localStorage.setItem(STORAGE_KEY, JSON.stringify(parsed));
   159→    return parsed;
   160→  } catch {
   161→    return {};
   162→  }
   163→}
   164→
   165→function saveSlots(slots: WeekSlots) {
   166→  if (typeof window === 'undefined') return;
   167→  localStorage.setItem(STORAGE_KEY, JSON.stringify(slots));
