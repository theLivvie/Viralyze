   125→// --- Animated Counter Component ---
   126→function AnimatedCounter({ value, duration = 1.2 }: { value: number; duration?: number }) {
   127→  const [display, setDisplay] = useState(0);
   128→  const ref = useRef(false);
   129→  const mv = useMotionValue(0);
   130→  const displayValue = useTransform(mv, (v) => Math.round(v));
   131→
   132→  useEffect(() => {
   133→    const unsub = displayValue.on('change', (v) => setDisplay(v));
   134→    return unsub;
   135→  }, [displayValue]);
   136→
   137→  useEffect(() => {
   138→    if (ref.current) return;
   139→    ref.current = true;
   140→    const ctrl = animate(mv, value, { duration, ease: 'easeOut' });
   141→    return () => { ctrl.stop(); };
   142→  }, [value, duration]);
   143→
   144→  return <>{display}</>;
   145→}
