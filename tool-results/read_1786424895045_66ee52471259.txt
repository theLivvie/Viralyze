    65→function getWeekDays(weekOffset: number): Date[] {
    66→  const now = new Date();
    67→  const currentDay = now.getDay();
    68→  const mondayOffset = currentDay === 0 ? -6 : 1 - currentDay;
    69→  const monday = new Date(now);
    70→  monday.setDate(now.getDate() + mondayOffset + weekOffset * 7);
    71→  monday.setHours(0, 0, 0, 0);
    72→
    73→  const days: Date[] = [];
    74→  for (let i = 0; i < 7; i++) {
    75→    const d = new Date(monday);
    76→    d.setDate(monday.getDate() + i);
    77→    days.push(d);
    78→  }
    79→  return days;
    80→}

    82→function dateKey(d: Date): string {
    83→  return d.toISOString().split('T')[0];

    86→function formatDayLabel(d: Date): string {
    87→  const today = new Date();
    88→  today.setHours(0, 0, 0, 0);
    89→  const target = new Date(d);
    90→  target.setHours(0, 0, 0, 0);
    91→  const diff = Math.round((target.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
    92→  if (diff === 0) return 'Today';
    93→  if (diff === 1) return 'Tomorrow';
    94→  if (diff === -1) return 'Yesterday';
    95→  return d.toLocaleDateString('en-US', { weekday: 'short', month: 'short', day: 'numeric' });
    96→}

   108→function isToday(d: Date): boolean {
   109→  const today = new Date();
   110→  return d.getDate() === today.getDate() &&
   111→    d.getMonth() === today.getMonth() &&
   112→    d.getFullYear() === today.getFullYear();
   113→}

   115→function loadSlots(): WeekSlots {
   116→  if (typeof window === 'undefined') return {};
   117→  try {
   118→    const raw = localStorage.getItem(STORAGE_KEY);
   119→    if (!raw) return {};
   120→    const parsed = JSON.parse(raw);
   121→    // Migrate old slots that may not have contentType
   122→    let needsSave = false;
   123→    for (const dk of Object.keys(parsed)) {
   124→      const updated = parsed[dk].map((slot: CalendarSlot) => {
   125→        if (!slot.contentType) { needsSave = true; return { ...slot, contentType: 'post' as ContentType }; }
   126→        return slot;
   127→      });
   128→      parsed[dk] = updated;
   129→    }
   130→    if (needsSave) localStorage.setItem(STORAGE_KEY, JSON.stringify(parsed));
   131→    return parsed;
   132→  } catch {
   133→    return {};
   134→  }
   135→}

   137→function saveSlots(slots: WeekSlots) {
   138→  if (typeof window === 'undefined') return;
   139→  localStorage.setItem(STORAGE_KEY, JSON.stringify(slots));
