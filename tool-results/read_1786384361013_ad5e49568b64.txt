   126→// --- Animated Counter Component ---
   127→function AnimatedCounter({ value, duration = 1.2 }: { value: number; duration?: number }) {
   128→  const [display, setDisplay] = useState(0);
   129→  const ref = useRef(false);
   130→  const mv = useMotionValue(0);
   131→  const displayValue = useTransform(mv, (v) => Math.round(v));
   132→
   133→  useEffect(() => {
   134→    const unsub = displayValue.on('change', (v) => setDisplay(v));
   135→    return unsub;
   136→  }, [displayValue]);
   137→
   138→  useEffect(() => {
   139→    if (ref.current) return;
   140→    ref.current = true;
   141→    const ctrl = animate(mv, value, { duration, ease: 'easeOut' });
   142→    return () => { ctrl.stop(); };
   143→  }, [value, duration]);
   144→
   145→  return <>{display}</>;
   146→}
