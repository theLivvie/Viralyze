   106→function CompareModal({
   107→  analyses: selected,
   108→  open,
   109→  onOpenChange,
   110→}: {
   111→  analyses: SavedAnalysis[];
   112→  open: boolean;
   113→  onOpenChange: (open: boolean) => void;
   114→}) {
   115→  const scoreColor = (score: number) => {
   116→    if (score >= 85) return 'text-emerald-400';
   117→    if (score >= 65) return 'text-green-400';
   118→    if (score >= 40) return 'text-amber-400';
   119→    return 'text-red-400';
   120→  };
   121→
   122→  const bestCategoryLabels: Record<string, string> = {
   123→    hook: 'Hook',
   124→    engagement: 'Engagement',
   125→    shareability: 'Shareability',
   126→    retention: 'Retention',
   127→    originality: 'Originality',
   128→    audienceFit: 'Audience Fit',
   129→    emotionalImpact: 'Emotional',
   130→    contentQuality: 'Quality',
   131→    trendAlignment: 'Trend',
   132→  };
   133→
   134→  // Compute best category per analysis
   135→  const getBestCategory = (a: SavedAnalysis) => {
   136→    if (!a.scores) return null;
   137→    const entries = Object.entries(a.scores).filter(([, v]) => typeof v === 'number');
   138→    if (entries.length === 0) return null;
   139→    entries.sort(([, a], [, b]) => (b as number) - (a as number));
   140→    const [key, value] = entries[0];
   141→    return { label: bestCategoryLabels[key] || key, value: value as number };
   142→  };
   143→
   144→  // Find winner IDs
   145→  const scoreWinnerId = selected.reduce<string | null>((
   146→    best,
   147→    a,
   148→  ) => {
   149→    const bestScore = selected.find((s) => s.id === best)?.overallScore ?? -1;
   150→    return a.overallScore > bestScore ? a.id : best;
   151→  }, selected[0]?.id ?? null);
   152→
   153→  const bestCategories = selected.map((a) => ({ id: a.id, best: getBestCategory(a) }));
   154→  const bestCategoryWinnerId = bestCategories.reduce<string | null>((
   155→    best,
   156→    curr,
   157→  ) => {
   158→    if (!curr.best) return best;
   159→    const bestVal = bestCategories.find((b) => b.id === best)?.best?.value ?? -1;
   160→    return curr.best.value > bestVal ? curr.id : best;
   161→  }, bestCategories[0]?.id ?? null);
   162→
   163→  return (
   164→    <Dialog open={open} onOpenChange={onOpenChange}>
   165→      <DialogContent className="bg-viralyze-black border-white/[0.08] max-w-3xl max-h-[80vh] overflow-y-auto">
   166→        <DialogHeader>
   167→          <DialogTitle className="text-viralyze-white flex items-center gap-2">
   168→            <GitCompareArrows className="h-5 w-5 text-wine-accent" />
   169→            Compare Analyses
   170→          </DialogTitle>
   171→          <DialogDescription className="text-viralyze-muted">
   172→            Side-by-side comparison of {selected.length} content analyses
   173→          </DialogDescription>
   174→        </DialogHeader>
   175→        <div className="mt-4">
   176→          <div className="overflow-x-auto">
   177→            <table className="w-full text-sm">
   178→              <thead>
   179→                <tr className="border-b border-white/[0.08]">
   180→                  <th className="text-left py-2.5 pr-4 text-viralyze-muted font-medium text-xs uppercase tracking-wider w-28">Field</th>
   181→                  {selected.map((a) => (
   182→                    <th key={a.id} className="text-left py-2.5 px-3 text-viralyze-white font-medium text-xs min-w-[180px]">
   183→                      {a.title || 'Untitled'}
   184→                    </th>
   185→                  ))}
   186→                </tr>
   187→              </thead>
   188→              <tbody className="divide-y divide-white/[0.04]">
   189→                {/* Score */}
   190→                <tr className="hover:glow-wine-sm transition-all">
   191→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Score</td>
   192→                  {selected.map((a) => (
   193→                    <td key={a.id} className="py-3 px-3">
   194→                      <span className={cn('text-lg font-bold tabular-nums', scoreColor(a.overallScore))}>
   195→                        {a.overallScore}
   196→                      </span>
   197→                      {scoreWinnerId === a.id && <span className="ml-1.5">🏆</span>}
   198→                    </td>
   199→                  ))}
   200→                </tr>
   201→                {/* Classification */}
   202→                <tr className="hover:glow-wine-sm transition-all">
   203→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Classification</td>
   204→                  {selected.map((a) => (
   205→                    <td key={a.id} className="py-3 px-3">
   206→                      <Badge
   207→                        variant="outline"
   208→                        className={cn('text-xs capitalize', classificationStyles[a.classification])}
   209→                      >
   210→                        {a.classification}
   211→                      </Badge>
   212→                    </td>
   213→                  ))}
   214→                </tr>
   215→                {/* Platform */}
   216→                <tr className="hover:glow-wine-sm transition-all">
   217→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Platform</td>
   218→                  {selected.map((a) => {
   219→                    const PI = platformIcons[a.platform];
   220→                    return (
   221→                      <td key={a.id} className="py-3 px-3">
   222→                        <div className="flex items-center gap-1.5 text-viralyze-white">
   223→                          <PI className="h-3.5 w-3.5 text-viralyze-muted" />
   224→                          <span className="text-xs capitalize">{a.platform}</span>
   225→                        </div>
   226→                      </td>
   227→                    );
   228→                  })}
   229→                </tr>
   230→                {/* Content Type */}
   231→                <tr className="hover:glow-wine-sm transition-all">
   232→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Content Type</td>
   233→                  {selected.map((a) => (
   234→                    <td key={a.id} className="py-3 px-3">
   235→                      <span className="text-xs text-viralyze-white capitalize">{a.contentType}</span>
   236→                    </td>
   237→                  ))}
   238→                </tr>
   239→                {/* Date */}
   240→                <tr className="hover:glow-wine-sm transition-all">
   241→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Date</td>
   242→                  {selected.map((a) => (
   243→                    <td key={a.id} className="py-3 px-3">
   244→                      <span className="text-xs text-viralyze-muted">
   245→                        {new Date(a.createdAt).toLocaleDateString('en-US', {
   246→                          month: 'short',
   247→                          day: 'numeric',
   248→                          year: 'numeric',
   249→                        })}
   250→                      </span>
   251→                    </td>
   252→                  ))}
   253→                </tr>
   254→                {/* Best Category */}
   255→                <tr className="hover:glow-wine-sm transition-all">
   256→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Best Category</td>
   257→                  {bestCategories.map((bc) => (
   258→                    <td key={bc.id} className="py-3 px-3">
   259→                      {bc.best ? (
   260→                        <div className="flex items-center gap-1.5">
   261→                          <span className="text-xs text-viralyze-white font-medium">{bc.best.label}</span>
   262→                          <span className={cn('text-xs font-bold tabular-nums', scoreColor(bc.best.value))}>
   263→                            {bc.best.value}
   264→                          </span>
   265→                          {bestCategoryWinnerId === bc.id && <span>🏆</span>}
   266→                        </div>
   267→                      ) : (
   268→                        <span className="text-xs text-viralyze-muted">N/A</span>
   269→                      )}
   270→                    </td>
   271→                  ))}
   272→                </tr>
   273→                {/* Visual score bar */}
   274→                <tr className="hover:glow-wine-sm transition-all">
   275→                  <td className="py-3 pr-4 text-viralyze-muted text-xs">Score Bar</td>
   276→                  {selected.map((a) => {
   277→                    const barColor = a.overallScore >= 70 ? 'bg-green-400' : a.overallScore >= 50 ? 'bg-amber-400' : 'bg-red-400';
   278→                    return (
   279→                      <td key={a.id} className="py-3 px-3">
   280→                        <div className="h-2.5 w-full max-w-[140px] rounded-full bg-white/[0.06] overflow-hidden">
   281→                          <div
   282→                            className={cn('h-full rounded-full transition-all duration-500', barColor)}
   283→                            style={{ width: `${a.overallScore}%` }}
   284→                          />
   285→                        </div>
   286→                      </td>
   287→                    );
   288→                  })}
   289→                </tr>
   290→              </tbody>
   291→            </table>
   292→          </div>
   293→        </div>
   294→      </DialogContent>
   295→    </Dialog>
   296→  );
   297→}

   299→export default function LibraryView() {
   300→  const { setSavedAnalyses, removeSavedAnalysis, setCurrentAnalysis, setCurrentView, user } = useAppStore();
   301→  const [analyses, setAnalyses] = useState<SavedAnalysis[]>([]);
   302→  const [loading, setLoading] = useState(true);
   303→  const [search, setSearch] = useState('');
   304→  const [platformFilter, setPlatformFilter] = useState<string>('all');
   305→  const [sort, setSort] = useState<string>('newest');
   306→  const [deleting, setDeleting] = useState<string | null>(null);
   307→  const [compareMode, setCompareMode] = useState(false);
   308→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   309→  const [compareOpen, setCompareOpen] = useState(false);
   310→
   311→  const fetchLibrary = useCallback(async () => {
   312→    if (!user?.id) {
   313→      setLoading(false);
   314→      return;
   315→    }
   316→    try {
   317→      setLoading(true);
   318→      const res = await fetch(`/api/library?userId=${user.id}`);
   319→      if (res.ok) {
   320→        const data = await res.json();
   321→        setAnalyses(data);
   322→        setSavedAnalyses(data);
   323→      }
   324→    } catch {
   325→      // Silently fail — use whatever is in local state
   326→    } finally {
   327→      setLoading(false);
   328→    }
   329→  }, [user?.id, setSavedAnalyses]);
   330→
   331→  useEffect(() => {
   332→    fetchLibrary();
   333→  }, [fetchLibrary]);
   334→
   335→  const filtered = analyses
   336→    .filter((a) => {
   337→      if (search && !a.title.toLowerCase().includes(search.toLowerCase())) return false;
   338→      if (platformFilter !== 'all' && a.platform !== platformFilter) return false;
   339→      return true;
   340→    })
   341→    .sort((a, b) => {
   342→      if (sort === 'newest') return new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
   343→      if (sort === 'oldest') return new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime();
   344→      if (sort === 'score-high') return b.overallScore - a.overallScore;
   345→      if (sort === 'score-low') return a.overallScore - b.overallScore;
   346→      return 0;
   347→    });
   348→
   349→  const handleDelete = async (e: React.MouseEvent, id: string) => {
   350→    e.stopPropagation();
   351→    if (!user?.id) return;
   352→    setDeleting(id);
   353→    try {
   354→      const res = await fetch(`/api/library?id=${id}&userId=${user.id}`, { method: 'DELETE' });
   355→      if (res.ok) {
   356→        setAnalyses((prev) => prev.filter((a) => a.id !== id));
   357→        removeSavedAnalysis(id);
   358→        setSelectedIds((prev) => { const next = new Set(prev); next.delete(id); return next; });
   359→        toast.success('Removed from library');
   360→      } else {
   361→        toast.error('Failed to delete analysis');
   362→      }
   363→    } catch {
   364→      toast.error('Failed to delete analysis');
   365→    } finally {
   366→      setDeleting(null);
   367→    }
   368→  };
   369→
   370→  const [viewingId, setViewingId] = useState<string | null>(null);
   371→
   372→  const handleCardClick = async (id: string) => {
   373→    if (compareMode) {
   374→      setSelectedIds((prev) => {
   375→        const next = new Set(prev);
   376→        if (next.has(id)) {
   377→          next.delete(id);
   378→        } else if (next.size < 3) {
   379→          next.add(id);
   380→        } else {
   381→          toast.info('You can compare up to 3 items');
   382→          return prev;
   383→        }
   384→        return next;
   385→      });
   386→    } else {
   387→      try {
   388→        setViewingId(id);
   389→        const res = await fetch(`/api/library?id=${id}`);
   390→        if (res.ok) {
   391→          const data = await res.json();
   392→          setCurrentAnalysis(data);
   393→          setCurrentView('analysis');
   394→        } else {
   395→          toast.error('Failed to load analysis');
   396→        }
   397→      } catch {
   398→        toast.error('Failed to load analysis');
   399→      } finally {
   400→        setViewingId(null);
   401→      }
   402→    }
   403→  };
   404→
   405→  const selectedAnalyses = analyses.filter((a) => selectedIds.has(a.id));
   406→
   407→  const handleCompare = () => {
   408→    if (selectedIds.size >= 2) {
   409→      setCompareOpen(true);
   410→    }
   411→  };
   412→
   413→  const exitCompareMode = () => {
   414→    setCompareMode(false);
   415→    setSelectedIds(new Set());
   416→  };
   417→
   418→  const isFilteredEmpty = !loading && filtered.length === 0 && analyses.length > 0;
   419→  const isEmpty = !loading && analyses.length === 0;
   420→
   421→  return (
   422→    <motion.div
   423→      variants={container}
   424→      initial="hidden"
   425→      animate="show"
   426→      className="flex flex-col gap-4 max-w-4xl mx-auto"
   427→    >
   428→      {/* Header with count badge */}
   429→      <motion.div variants={item} className="flex items-center gap-3">
   430→        <div className="flex flex-col">
   431→          <div className="flex items-center gap-2.5">
   432→            <h2 className="text-2xl md:text-3xl font-bold text-viralyze-white">
   433→              Content Library
   434→            </h2>
   435→            {!loading && analyses.length > 0 && (
   436→              <Badge
   437→                variant="outline"
   438→                className="text-xs border-wine-accent/30 text-wine-accent bg-wine-accent/10 tabular-nums"
   439→              >
   440→                {analyses.length}
   441→              </Badge>
   442→            )}
   443→          </div>
   444→          <p className="text-viralyze-muted mt-1">
   445→            Your saved content analyses
   446→          </p>
   447→        </div>
   448→      </motion.div>
   449→
   450→      {/* Filters */}
   451→      <motion.div variants={item} className="flex flex-col sm:flex-row gap-3">
   452→        <div className="relative flex-1">
   453→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-viralyze-muted" />
   454→          <Input
   455→            placeholder="Search analyses..."
   456→            value={search}
   457→            onChange={(e) => setSearch(e.target.value)}
   458→            className="pl-9 bg-white/[0.05] border-white/[0.08] text-viralyze-white placeholder:text-viralyze-muted/40 focus-visible:ring-wine-accent hover:border-white/[0.12] transition-colors"
   459→          />
   460→        </div>
   461→        <Select value={platformFilter} onValueChange={setPlatformFilter}>
   462→          <SelectTrigger className="w-full sm:w-40 bg-white/[0.05] border-white/[0.08] text-viralyze-white hover:border-white/[0.12] transition-colors">
   463→            <SelectValue placeholder="Platform" />
   464→          </SelectTrigger>
   465→          <SelectContent className="bg-viralyze-soft-black border-white/[0.08]">
   466→            <SelectItem value="all" className="text-viralyze-white">All Platforms</SelectItem>
   467→            <SelectItem value="instagram" className="text-viralyze-white">Instagram</SelectItem>
   468→            <SelectItem value="youtube" className="text-viralyze-white">YouTube</SelectItem>
   469→            <SelectItem value="tiktok" className="text-viralyze-white">TikTok</SelectItem>
   470→            <SelectItem value="x" className="text-viralyze-white">X</SelectItem>
   471→            <SelectItem value="linkedin" className="text-viralyze-white">LinkedIn</SelectItem>
   472→          </SelectContent>
   473→        </Select>
   474→        <Select value={sort} onValueChange={setSort}>
   475→          <SelectTrigger className="w-full sm:w-40 bg-white/[0.05] border-white/[0.08] text-viralyze-white hover:border-white/[0.12] transition-colors">
   476→            <SelectValue placeholder="Sort" />
   477→          </SelectTrigger>
   478→          <SelectContent className="bg-viralyze-soft-black border-white/[0.08]">
   479→            <SelectItem value="newest" className="text-viralyze-white">Newest First</SelectItem>
   480→            <SelectItem value="oldest" className="text-viralyze-white">Oldest First</SelectItem>
   481→            <SelectItem value="score-high" className="text-viralyze-white">Score: High to Low</SelectItem>
   482→            <SelectItem value="score-low" className="text-viralyze-white">Score: Low to High</SelectItem>
   483→          </SelectContent>
   484→        </Select>
   485→      </motion.div>
   486→
   487→      {/* Compare controls */}
   488→      {!loading && analyses.length > 1 && (
   489→        <motion.div variants={item} className="flex items-center gap-2">
   490→          <Button
   491→            variant={compareMode ? 'default' : 'outline'}
   492→            size="sm"
   493→            onClick={() => {
   494→              if (compareMode) {
   495→                exitCompareMode();
   496→              } else {
   497→                setCompareMode(true);
   498→              }
   499→            }}
   500→            className={cn(
   501→              'gap-1.5 transition-all',
   502→              compareMode
   503→                ? 'bg-wine-accent hover:bg-wine-accent/80 text-white border-wine-accent'
   504→                : 'border-white/[0.08] text-viralyze-white hover:bg-white/[0.05] hover:border-white/20'
   505→            )}
   506→          >
   507→            <GitCompareArrows className="h-3.5 w-3.5" />
   508→            {compareMode ? 'Exit Compare' : 'Compare'}
   509→          </Button>
   510→          {compareMode && selectedIds.size > 0 && (
   511→            <>
   512→              <span className="text-xs text-viralyze-muted">
   513→                {selectedIds.size}/3 selected
   514→              </span>
   515→              {selectedIds.size >= 2 && (
   516→                <Button
   517→                  size="sm"
   518→                  onClick={handleCompare}
   519→                  className="bg-wine-accent hover:bg-wine-accent/80 text-white gap-1.5"
   520→                >
   521→                  Compare Selected ({selectedIds.size})
   522→                </Button>
   523→              )}
   524→              <Button
   525→                size="sm"
   526→                variant="ghost"
   527→                onClick={() => setSelectedIds(new Set())}
   528→                className="text-viralyze-muted hover:text-viralyze-white gap-1"
   529→              >
   530→                <X className="h-3 w-3" />
   531→                Clear
   532→              </Button>
   533→            </>
   534→          )}
   535→        </motion.div>
   536→      )}
   537→
   538→      {/* Inline Comparison Panel — shows when exactly 2 items selected */
   539→      <AnimatePresence>
