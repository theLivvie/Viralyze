   143→function HeatBar({ heat }: { heat: number }) {
   144→  const heatColor = heat >= 5 ? 'from-wine-accent via-orange-400 to-red-500' : heat >= 4 ? 'from-wine-accent via-orange-400 to-wine-accent' : heat >= 3 ? 'from-wine-accent/80 via-amber-400 to-wine-accent/80' : 'from-wine-accent/50 via-amber-400/50 to-wine-accent/50';
   145→  return (
   146→    <div className="h-1.5 rounded-full bg-white/[0.06] w-full overflow-hidden relative">
   147→      {/* Heat level segments (1-5) */}
   148→      <div className="absolute inset-0 flex">
   149→        {Array.from({ length: 5 }).map((_, i) => (
   150→          <div key={i} className="flex-1 border-r border-white/[0.04] last:border-r-0" />
   151→        ))}
   152→      </div>
   153→      <motion.div
   154→        className={cn('h-full rounded-full bg-gradient-to-r', heatColor)}
   155→        initial={{ width: 0 }}
   156→        animate={{ width: `${(heat / 5) * 100}%` }}
   157→        transition={{ duration: 0.8, ease: 'easeOut', delay: 0.2 }}
   158→      />
   159→      {/* Glow at the tip */}
   160→      {heat >= 4 && (
   161→        <motion.div
   162→          className="absolute top-1/2 -translate-y-1/2 h-2.5 w-2.5 rounded-full bg-orange-400/40 blur-sm pointer-events-none"
   163→          initial={{ opacity: 0, left: 0 }}
   164→          animate={{ opacity: [0.4, 0.8, 0.4], left: `${(heat / 5) * 100}%`, x: '-5px' }}
   165→          transition={{ duration: 0.8, ease: 'easeOut', delay: 0.2, opacity: { repeat: Infinity, duration: 1.5 } }}
   166→        />
   167→      )}
   168→    </div>
   169→  );
   170→}

   225→    setBookmarkedTrends((prev) => {
   226→      const next = new Set(prev);
   227→      if (next.has(trendName)) {
   228→        next.delete(trendName);
   229→        toast.info('Bookmark removed');
   230→      } else {
   231→        next.add(trendName);
   232→        toast.success('Trend bookmarked!');
   233→      }
   234→      return next;
   235→    });
   236→  };
   237→
   238→  const clearFilters = () => {
   239→    setSearchQuery('');

   241→  };
   242→
   243→  // Live clock tick
   244→  useEffect(() => {
   245→    const interval = setInterval(() => setLiveClock(new Date()), 1000);
   246→    return () => clearInterval(interval);
   247→  }, []);
   248→
   249→  const fetchTrends = useCallback(async () => {
   250→    setLoading(true);
   251→    setError(false);
   252→    try {
   253→      const res = await fetch('/api/trends');
   254→      if (!res.ok) throw new Error('Failed to fetch');
   255→      const data = await res.json();
   256→      if (data.categories && Array.isArray(data.categories)) {
   257→        const mapped: TrendCategory[] = data.categories.map(
   258→          (cat: { category: string; trends: { name: string; growth: string; heat: number; platforms: string[]; label?: string }[] }) => ({
   259→            category: cat.category,
   260→            icon: categoryIconMap[cat.category.toLowerCase()] || TrendingUp,
   261→            trends: cat.trends.map((t) => ({
   262→              name: t.name,
   263→              heat: t.heat,
   264→              growth: t.growth,
   265→              platforms: (t.platforms || []).filter((p: string) => p in platformIcons) as Platform[],
   266→              category: cat.category.toLowerCase(),
   267→              label: t.label,
   268→            })),
   269→          })
   270→        );
   271→        setTrendData(mapped);
   272→        setLastUpdated(new Date());
   273→      } else {
   274→        setTrendData(fallbackTrendData);
   275→      }
   276→    } catch {
   277→      setError(true);
   278→    } finally {
   279→      setLoading(false);
   280→    }
   281→  }, []);
   282→
   283→  // Filtered trend data based on search and category filter
   284→  const filteredTrendData = useMemo(() => {
   285→    let data = trendData;
   286→
   287→    // Apply category filter
   288→    if (activeFilter !== 'All') {
   289→      const matchCategories = filterToDataCategory[activeFilter] || [];
   290→      if (matchCategories.length > 0) {
   291→        data = data.filter((cat) =>
   292→          matchCategories.some((mc) => cat.category.toLowerCase() === mc)
   293→        );
   294→      }
   295→    }
   296→
   297→    // Apply search query
   298→    if (searchQuery.trim()) {
   299→      const q = searchQuery.toLowerCase().trim();
   300→      data = data
   301→        .map((cat) => ({
   302→          ...cat,
   303→          trends: cat.trends.filter((t) => t.name.toLowerCase().includes(q)),
   304→        }))
   305→        .filter((cat) => cat.trends.length > 0);
   306→    }
   307→
   308→    return data;
   309→  }, [trendData, activeFilter, searchQuery]);
   310→
   311→  const hasAnyResults = filteredTrendData.some((cat) => cat.trends.length > 0);
   312→

   316→      initial="hidden"

   319→    >
   320→      {/* Header with refresh and last updated — live clock feel */}
   321→      <motion.div variants={item} className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   322→        <div className="flex flex-col">
   323→          <div className="flex items-center gap-2">
   324→            <p className="text-viralyze-muted text-sm">
   325→              Real-time trending topics across platforms

   327→            <div className="flex items-center gap-1 text-viralyze-muted/30">
   328→              <Clock className="h-3 w-3" />
   329→              <span className="text-[10px] font-mono tabular-nums">
   330→                {liveClock.toLocaleTimeString('en-US', { hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' })}
   331→              </span>
   332→            </div>
   333→            {lastUpdated && (

   353→          {loading ? (
   354→            <Loader2 className="h-3.5 w-3.5 animate-spin" />

   356→            <RefreshCw className="h-3.5 w-3.5" />
