   135→// --- Animated Counter Component ---
   136→function AnimatedCounter({ value, duration = 1.2 }: { value: number; duration?: number }) {
   137→  const [display, setDisplay] = useState(0);
   138→  const ref = useRef(false);
   139→  const mv = useMotionValue(0);
   140→  const displayValue = useTransform(mv, (v) => Math.round(v));
   141→
   142→  useEffect(() => {
   143→    const unsub = displayValue.on('change', (v) => setDisplay(v));
   144→    return unsub;
   145→  }, [displayValue]);
   146→
   147→  useEffect(() => {
   148→    if (ref.current) return;
   149→    ref.current = true;
   150→    const ctrl = animate(mv, value, { duration, ease: 'easeOut' });
   151→    return () => { ctrl.stop(); };
   152→  }, [value, duration]);
   153→
   154→  return <>{display}</>;
   155→}
